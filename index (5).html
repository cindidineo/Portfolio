/* ---------- Utilities ---------- */
function uuidv4(){
  // simple UUID generator for demo purposes
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, c => {
    const r = Math.random() * 16 | 0;
    return (c === 'x' ? r : (r & 0x3 | 0x8)).toString(16);
  });
}
function escapeHtml(s){
  if(!s) return '';
  return s.replaceAll('&','&').replaceAll('<','<').replaceAll('>','>').replaceAll('"','"');
}

/* ---------- Local storage helpers ---------- */
const LS_USERS_KEY = 'nexora_users_v1';
const LS_CURRENT_KEY = 'nexora_current_user_v1';
const LS_TASKS_KEY = 'nexora_tasks_v1';

function getUsers(){
  try { return JSON.parse(localStorage.getItem(LS_USERS_KEY) || '[]'); } catch(e){ return []; }
}
function saveUsers(u){ localStorage.setItem(LS_USERS_KEY, JSON.stringify(u || [])); }
function setCurrentUser(u){ localStorage.setItem(LS_CURRENT_KEY, JSON.stringify(u || null)); }
function getCurrentUser(){ try { return JSON.parse(localStorage.getItem(LS_CURRENT_KEY) || 'null'); } catch(e){ return null; } }
function getAllTasks(){ try { return JSON.parse(localStorage.getItem(LS_TASKS_KEY) || '[]'); } catch(e){ return []; } }
function saveAllTasks(t){ localStorage.setItem(LS_TASKS_KEY, JSON.stringify(t || [])); }

/* Initialize sample data if none exists */
if(!localStorage.getItem(LS_TASKS_KEY)){
  saveAllTasks(SAMPLE_TASKS);
  document.getElementById('json-sample').innerText = JSON.stringify(SAMPLE_TASKS, null, 2);
} else {
  document.getElementById('json-sample').innerText = JSON.stringify(getAllTasks(), null, 2);
}

/* ---------- Auth (local / supabase) ---------- */
async function signUp(){
  const email = emailInput.value.trim();
  const pw = passwordInput.value.trim();
  if(!email || !pw){ authMsg.innerText = 'Email and password required'; return; }
  if(useSupabase && supabase){
    const { data, error } = await supabase.auth.signUp({ email, password: pw });
    if(error) authMsg.innerText = error.message;
    else authMsg.innerText = 'Sign-up sent: check your email (Supabase)';
  } else {
    const users = getUsers();
    if(users.find(u=>u.email===email)){ authMsg.innerText = 'User already exists (local)'; return; }
    const user = { id: uuidv4(), email, password: pw };
    users.push(user);
    saveUsers(users);
    setCurrentUser({ id: user.id, email: user.email });
    authMsg.innerText = 'Signed up (local)';
    loadApp({ id: user.id, email: user.email });
  }
}
async function signIn(){
  const email = emailInput.value.trim();
  const pw = passwordInput.value.trim();
  if(!email || !pw){ authMsg.innerText = 'Email and password required'; return; }
  if(useSupabase && supabase){
    const { data, error } = await supabase.auth.signInWithPassword({ email, password: pw });
    if(error) authMsg.innerText = error.message;
    else loadApp(data.user);
  } else {
    const users = getUsers();
    const user = users.find(u=>u.email===email && u.password===pw);
    if(!user){ authMsg.innerText = 'Invalid credentials (local)'; return; }
    setCurrentUser({ id: user.id, email: user.email });
    authMsg.innerText = 'Signed in (local)';
    loadApp({ id: user.id, email: user.email });
  }
}
async function logout(){
  if(useSupabase && supabase){
    await supabase.auth.signOut();
  }
  setCurrentUser(null);
  currentUser = null;
  appBox.classList.add('hidden');
  authBox.classList.remove('hidden');
}

/* ---------- CRUD operations (local / supabase) ---------- */
let currentUser = null;

async function loadApp(user){
  currentUser = user;
  authBox.classList.add('hidden');
  appBox.classList.remove('hidden');
  fetchTasks();
}

async function addTask(){
  const title = titleInput.value.trim();
  const desc = descInput.value.trim();
  const status = statusSelect.value;
  if(!title){ alert('Title required'); return; }

  if(useSupabase && supabase){
    const payload = { title, description: desc, status, user_id: currentUser.id };
    const { data, error } = await supabase.from('tasks').insert([payload]);
    if(error) return alert('Error creating task: ' + error.message);
  } else {
    const tasks = getAllTasks();
    const t = { id: uuidv4(), user_id: currentUser.id, title, description: desc, status, created_at: new Date().toISOString() };
    tasks.unshift(t);
    saveAllTasks(tasks);
    document.getElementById('json-sample').innerText = JSON.stringify(tasks, null, 2);
  }
  titleInput.value = '';
  descInput.value = '';
  fetchTasks();
}

async function fetchTasks(){
  if(useSupabase && supabase){
    const { data, error } = await supabase.from('tasks').select('*').eq('user_id', currentUser.id).order('created_at', { ascending: false });
    if(error) { tasksDiv.innerHTML = '
Error loading tasks: '+escapeHtml(error.message)+'
'; return; }
    renderTasks(data || []);
  } else {
    const tasks = getAllTasks().filter(t => t.user_id === currentUser.id);
    tasks.sort((a,b) => new Date(b.created_at) - new Date(a.created_at));
    renderTasks(tasks);
  }
}

function renderTasks(list){
  if(!list.length){ tasksDiv.innerHTML = '
No tasks yet. Add one above.
'; return; }
  tasksDiv.innerHTML = list.map(t => {
    const title = escapeHtml(t.title);
    const desc = escapeHtml(t.description || '');
    const when = new Date(t.created_at || Date.now()).toLocaleString();
    const id = t.id;
    const status = t.status || 'pending';
    return `
      

        

          

            
${title}

            
${escapeHtml(status)} · ${when}

          

          

            
            
          

        

        
${desc}

      

    `;
  }).join('');
}

/* Edit task (prompt-based for simplicity) */
async function editTask(id){
  const newTitle = prompt('New title:');
  if(newTitle === null) return;
  const newDesc = prompt('New description (leave blank to keep):', '') ?? '';
  const newStatus = prompt('New status (pending, in-progress, done):', 'pending') || 'pending';

  if(useSupabase && supabase){
    const { error } = await supabase.from('tasks').update({ title: newTitle, description: newDesc, status: newStatus }).eq('id', id);
    if(error) return alert('Update error: ' + error.message);
  } else {
    const tasks = getAllTasks();
    const idx = tasks.findIndex(t => t.id === id && t.user_id === currentUser.id);
    if(idx === -1) return alert('Task not found');
    tasks[idx].title = newTitle;
    if(newDesc) tasks[idx].description = newDesc;
    tasks[idx].status = newStatus;
    saveAllTasks(tasks);
    document.getElementById('json-sample').innerText = JSON.stringify(tasks, null, 2);
  }
  fetchTasks();
}

async function deleteTask(id){
  if(!confirm('Delete this task?')) return;
  if(useSupabase && supabase){
    const { error } = await supabase.from('tasks').delete().eq('id', id);
    if(error) return alert('Delete error: ' + error.message);
  } else {
    let tasks = getAllTasks();
    tasks = tasks.filter(t => !(t.id === id && t.user_id === currentUser.id));
    saveAllTasks(tasks);
    document.getElementById('json-sample').innerText = JSON.stringify(tasks, null, 2);
  }
  fetchTasks();
}

createBtn.addEventListener('click', () => addTask());
exportJsonBtn.addEventListener('click', () => {
  const tasks = getAllTasks();
  const blob = new Blob([JSON.stringify(tasks, null, 2)], {type: 'application/json'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = 'tasks.json';
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
});

/* ---------- On load: check logged-in state ---------- */
(async function init(){
  if(useSupabase && supabase){
    // Try to get supabase user
    try {
      const { data } = await supabase.auth.getUser();
      if(data && data.user){ loadApp(data.user); return; }
    } catch(e){}
  } else {
    const u = getCurrentUser();
    if(u){ loadApp(u); return; }
  }
  // otherwise stay on auth screen
})();
